### 3.1 `src/lib/api.ts` — Axios实例配置

```typescript
// 这是整个前端HTTP请求的核心文件
// 所有发往后端的请求都经过这里，自动带上Token、统一处理错误

import axios from 'axios'
import type { ApiResponse } from '@/types/api'

// 创建Axios实例
const api = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_BASE_URL || '/api/v1',
  timeout: 30000, // 请求超时时间：30秒
  headers: {
    'Content-Type': 'application/json',
  },
})

// ========== 请求拦截器 ==========
// 每次发送请求前自动执行
api.interceptors.request.use(
  (config) => {
    // 从浏览器本地存储中获取Token
    const token = localStorage.getItem('token')
    if (token) {
      // 在请求头中加上 Authorization: Bearer xxx
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// ========== 响应拦截器 ==========
// 每次收到后端响应后自动执行
api.interceptors.response.use(
  (response) => {
    const data = response.data as ApiResponse

    // 如果业务码不为0，说明有业务错误
    if (data.code !== 0) {
      // Token过期或未登录 → 跳转到登录页
      if (data.code === 40101) {
        localStorage.removeItem('token')
        localStorage.removeItem('user')
        // 避免在登录页循环跳转
        if (typeof window !== 'undefined' && !window.location.pathname.includes('/login')) {
          window.location.href = '/login'
        }
      }
      // 抛出错误，让调用方catch处理
      return Promise.reject(new Error(data.message || '请求失败'))
    }

    return response
  },
  (error) => {
    // 网络错误等
    if (error.response) {
      switch (error.response.status) {
        case 401:
          localStorage.removeItem('token')
          if (typeof window !== 'undefined') {
            window.location.href = '/login'
          }
          break
        case 403:
          console.error('权限不足')
          break
        case 500:
          console.error('服务器错误')
          break
      }
    } else {
      console.error('网络异常，请检查网络连接')
    }
    return Promise.reject(error)
  }
)

export default api
```

```typescript
// 日期格式化
export function formatDate(date: string | Date, format = 'YYYY-MM-DD HH:mm:ss'): string {
  if (!date) return '-'
  return dayjs(date).format(format)
}

// 文件大小格式化（字节 → 人类可读）
export function formatFileSize(bytes: number): string {
  if (bytes === 0) return '0 B'
  const units = ['B', 'KB', 'MB', 'GB']
  const i = Math.floor(Math.log(bytes) / Math.log(1024))
  return (bytes / Math.pow(1024, i)).toFixed(2) + ' ' + units[i]
}

  return status === 1 ? '启用' : '停用'
```

## 4. 全局状态管理（Context）

### `src/contexts/AuthContext.tsx`

```tsx
'use client'

// AuthContext 管理全局的登录状态
// 通俗说：让所有页面都能知道"当前是谁登录了"

import React, { createContext, useContext, useState, useEffect, useCallback } from 'react'
import { useRouter } from 'next/navigation'
import type { User } from '@/types/user'
import authService from '@/services/authService'

// 定义Context的类型
interface AuthContextType {
  user: User | null          // 当前登录用户，null表示未登录
  loading: boolean           // 是否正在加载用户信息
  login: (token: string, user: User) => void  // 登录
  logout: () => void         // 登出
  isAdmin: boolean           // 是否管理员
  isOperator: boolean        // 是否操作员
  hasWritePermission: boolean // 是否有写权限
}

const AuthContext = createContext<AuthContextType | undefined>(undefined)

// AuthProvider 包裹在最外层，给所有子组件提供认证信息
export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null)
  const [loading, setLoading] = useState(true)
  const router = useRouter()

  // 页面加载时，检查是否已经登录（从localStorage恢复登录状态）
  useEffect(() => {
    const token = localStorage.getItem('token')
    const savedUser = localStorage.getItem('user')

    if (token && savedUser) {
      try {
        setUser(JSON.parse(savedUser))
      } catch {
        localStorage.removeItem('token')
        localStorage.removeItem('user')
      }
    }
    setLoading(false)
  }, [])

  // 登录：保存Token和用户信息
  const login = useCallback((token: string, userData: User) => {
    localStorage.setItem('token', token)
    localStorage.setItem('user', JSON.stringify(userData))
    setUser(userData)
  }, [])

  // 登出：清除Token和用户信息，跳转登录页
  const logout = useCallback(async () => {
    try {
      await authService.logout()
    } catch {
      // 即使登出接口失败也要清除本地状态
    }
    localStorage.removeItem('token')
    localStorage.removeItem('user')
    setUser(null)
    router.push('/login')
  }, [router])

  const value: AuthContextType = {
    user,
    loading,
    login,
    logout,
    isAdmin: user?.role === 'admin',
    isOperator: user?.role === 'operator',
    hasWritePermission: user?.role === 'admin' || user?.role === 'operator',
  }

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>
}

// useAuth Hook：在任何组件中获取认证信息
// 用法：const { user, logout, isAdmin } = useAuth()
export function useAuth() {
  const context = useContext(AuthContext)
  if (!context) {
    throw new Error('useAuth 必须在 AuthProvider 内部使用')
  }
  return context
}
```

### 5.1 `src/hooks/useInventory.ts`

```typescript
'use client'

// 封装库存数据的查询、缓存、新增、编辑、删除操作
// 使用 React Query，自动管理加载状态、缓存、刷新

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { message } from 'antd'
import inventoryService from '@/services/inventoryService'
import type { InventoryQuery, CreateInventoryRequest, UpdateInventoryRequest } from '@/types/inventory'

// 查询库存列表
export function useInventoryList(params: InventoryQuery) {
  return useQuery({
    queryKey: ['inventories', params],        // 缓存Key（参数变化自动重新查询）
    queryFn: () => inventoryService.getList(params), // 实际查询函数
  })
}

// 查询库存详情
export function useInventoryDetail(id: number) {
  return useQuery({
    queryKey: ['inventory', id],
    queryFn: () => inventoryService.getById(id),
    enabled: id > 0, // 只有ID有效时才查询
  })
}

// 新增库存
export function useCreateInventory() {
  const queryClient = useQueryClient()

  return useMutation({
    mutationFn: (data: CreateInventoryRequest) => inventoryService.create(data),
    onSuccess: () => {
      message.success('新增成功')
      queryClient.invalidateQueries({ queryKey: ['inventories'] }) // 刷新列表缓存
    },
    onError: (error: Error) => {
      message.error(error.message || '新增失败')
    },
  })
}

// 编辑库存
export function useUpdateInventory() {
  const queryClient = useQueryClient()

  return useMutation({
    mutationFn: ({ id, data }: { id: number; data: UpdateInventoryRequest }) =>
      inventoryService.update(id, data),
    onSuccess: () => {
      message.success('编辑成功')
      queryClient.invalidateQueries({ queryKey: ['inventories'] })
      queryClient.invalidateQueries({ queryKey: ['inventory'] })
    },
    onError: (error: Error) => {
      message.error(error.message || '编辑失败')
    },
  })
}

// 删除库存
export function useDeleteInventory() {
  const queryClient = useQueryClient()

  return useMutation({
    mutationFn: (id: number) => inventoryService.delete(id),
    onSuccess: () => {
      message.success('删除成功')
      queryClient.invalidateQueries({ queryKey: ['inventories'] })
    },
    onError: (error: Error) => {
      message.error(error.message || '删除失败')
    },
  })
}
```

```tsx
'use client'

// 文件上传组件：支持拖拽上传 + 点击选择

import { Upload, message } from 'antd'
import { InboxOutlined } from '@ant-design/icons'
import { MAX_FILE_SIZE, ALLOWED_UPLOAD_TYPES } from '@/lib/constants'
import { formatFileSize } from '@/lib/utils'

const { Dragger } = Upload

interface Props {
  inventoryId: number
  onSuccess: () => void // 上传成功后回调（刷新附件列表）
}

export default function FileUploader({ inventoryId, onSuccess }: Props) {
  const token = typeof window !== 'undefined' ? localStorage.getItem('token') : ''
  const baseUrl = process.env.NEXT_PUBLIC_API_BASE_URL || '/api/v1'

  return (
    <Dragger
      name="file"
      multiple
      action={`${baseUrl}/inventories/${inventoryId}/attachments`}
      headers={{ Authorization: `Bearer ${token}` }}
      beforeUpload={(file) => {
        // 检查文件大小
        if (file.size > MAX_FILE_SIZE) {
          message.error(`${file.name} 超过大小限制（最大${formatFileSize(MAX_FILE_SIZE)}）`)
          return Upload.LIST_IGNORE
        }
        // 检查文件格式
        const ext = '.' + file.name.split('.').pop()?.toLowerCase()
        if (!ALLOWED_UPLOAD_TYPES.includes(ext)) {
          message.error(`${file.name} 格式不支持`)
          return Upload.LIST_IGNORE
        }
        return true
      }}
      onChange={(info) => {
        if (info.file.status === 'done') {
          message.success(`${info.file.name} 上传成功`)
          onSuccess()
        } else if (info.file.status === 'error') {
          message.error(`${info.file.name} 上传失败`)
        }
      }}
    >
      <p className="text-4xl text-gray-400">
        <InboxOutlined />
      </p>
      <p className="text-base mt-2">点击或拖拽文件到此区域上传</p>
      <p className="text-sm text-gray-400">
        支持 jpg/png/pdf/doc/xls 等格式，单文件最大 20MB
      </p>
    </Dragger>
  )
}
```

```tsx
'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import { Button, Space } from 'antd'
import { PlusOutlined, ExportOutlined, DeleteOutlined } from '@ant-design/icons'
import { saveAs } from 'file-saver'
import PageTitle from '@/components/common/PageTitle'
import InventoryFilter from '@/components/inventory/InventoryFilter'
import InventoryTable from '@/components/inventory/InventoryTable'
import { showConfirm } from '@/components/common/ConfirmModal'
import { useAuth } from '@/contexts/AuthContext'
import { useInventoryList, useDeleteInventory, useBatchDeleteInventory } from '@/hooks/useInventory'
import inventoryService from '@/services/inventoryService'
import type { InventoryQuery } from '@/types/inventory'

export default function AdminInventoryPage() {
  const router = useRouter()
  const { isAdmin, hasWritePermission } = useAuth()

  // 查询参数状态
  const [query, setQuery] = useState<InventoryQuery>({ page: 1, page_size: 20 })
  const [selectedIds, setSelectedIds] = useState<number[]>([])

  // 数据查询
  const { data, isLoading } = useInventoryList(query)

  // 删除操作
  const deleteMutation = useDeleteInventory()
  const batchDeleteMutation = useBatchDeleteInventory()

  // 筛选
  const handleFilter = (values: Partial<InventoryQuery>) => {
    setQuery({ ...values, page: 1, page_size: query.page_size })
  }

  // 翻页
  const handlePageChange = (page: number, pageSize: number) => {
    setQuery({ ...query, page, page_size: pageSize })
  }

  // 单条删除
  const handleDelete = (id: number) => {
    showConfirm('确认删除', '确定要删除该库存记录？删除后可在"已删除数据"中恢复。', () => {
      deleteMutation.mutate(id)
    })
  }

  // 批量删除
  const handleBatchDelete = () => {
    if (selectedIds.length === 0) return
    showConfirm('批量删除', `确定要删除选中的 ${selectedIds.length} 条记录？`, () => {
      batchDeleteMutation.mutate(selectedIds, {
        onSuccess: () => setSelectedIds([]),
      })
    })
  }

  // 导出
  const handleExport = async () => {
    try {
      const blob = await inventoryService.export(query)
      saveAs(blob, `库存数据_${new Date().toISOString().slice(0, 10)}.xlsx`)
    } catch {
      // 错误已在拦截器中处理
    }
  }

  return (
    <>
      <PageTitle
        title="库存管理"
        extra={
          <Space>
            <Button icon={<ExportOutlined />} onClick={handleExport}>导出</Button>
            {hasWritePermission && (
              <Button type="primary" icon={<PlusOutlined />}
                onClick={() => router.push('/admin/inventory/create')}>
                新增库存
              </Button>
            )}
          </Space>
        }
      />

      <InventoryFilter onFilter={handleFilter} loading={isLoading} />

      {/* 批量操作栏 */}
      {selectedIds.length > 0 && isAdmin && (
        <div className="mb-4 p-3 bg-blue-50 rounded flex items-center gap-4">
          <span>已选中 {selectedIds.length} 条</span>
          <Button danger icon={<DeleteOutlined />} onClick={handleBatchDelete}
            loading={batchDeleteMutation.isPending}>
            批量删除
          </Button>
        </div>
      )}

      <InventoryTable
        data={data?.list || []}
        total={data?.total || 0}
        page={query.page || 1}
        pageSize={query.page_size || 20}
        loading={isLoading}
        showEdit={hasWritePermission}
        showDelete={isAdmin}
        selectedRowKeys={selectedIds}
        onPageChange={handlePageChange}
        onView={(id) => router.push(`/inventory/${id}`)}
        onEdit={(id) => router.push(`/admin/inventory/${id}/edit`)}
        onDelete={handleDelete}
        onSelectChange={isAdmin ? setSelectedIds : undefined}
      />
    </>
  )
}
```

### 8.7 `src/app/admin/inventory/[id]/edit/page.tsx` — 编辑库存页

```tsx
'use client'

import { useParams, useRouter } from 'next/navigation'
import { Spin } from 'antd'
import PageTitle from '@/components/common/PageTitle'
import InventoryForm from '@/components/inventory/InventoryForm'
import { useInventoryDetail, useUpdateInventory } from '@/hooks/useInventory'

export default function EditInventoryPage() {
  const { id } = useParams<{ id: string }>()
  const router = useRouter()

  const { data: inventory, isLoading } = useInventoryDetail(Number(id))
  const updateMutation = useUpdateInventory()

  if (isLoading) return <Spin className="flex justify-center mt-20" size="large" />
  if (!inventory) return <div className="text-center mt-20">库存记录不存在</div>

  return (
    <>
      <PageTitle title={`编辑库存 - ${inventory.material_code}`} />
      <InventoryForm
        isEdit
        initialValues={inventory}
        onSubmit={(values) => {
          updateMutation.mutate(
            { id: Number(id), data: { ...values, version: inventory.version } },
            { onSuccess: () => router.push('/admin/inventory') }
          )
        }}
        onCancel={() => router.back()}
        loading={updateMutation.isPending}
      />
    </>
  )
}
```

### 8.8 `src/app/inventory/[id]/page.tsx` — 库存详情页（前台）

```tsx
'use client'

import { useParams, useRouter } from 'next/navigation'
import { Button, Card, Descriptions, Tag, Image, Table, Spin, Divider } from 'antd'
import { ArrowLeftOutlined, DownloadOutlined, WarningFilled } from '@ant-design/icons'
import { useInventoryDetail } from '@/hooks/useInventory'
import { CATEGORY_MAP } from '@/types/inventory'
import { formatDate, formatFileSize, getStatusText } from '@/lib/utils'
import attachmentService from '@/services/attachmentService'
import type { Attachment } from '@/types/attachment'

export default function InventoryDetailPage() {
  const { id } = useParams<{ id: string }>()
  const router = useRouter()
  const { data: inv, isLoading } = useInventoryDetail(Number(id))

  if (isLoading) return <Spin className="flex justify-center mt-20" size="large" />
  if (!inv) return <div className="text-center mt-20">记录不存在</div>

  // 分离图片和文件
  const images = inv.attachments?.filter(a => a.file_type === 'image') || []
  const documents = inv.attachments?.filter(a => a.file_type === 'document') || []

  return (
    <div className="max-w-5xl mx-auto p-6">
      <Button icon={<ArrowLeftOutlined />} onClick={() => router.back()} className="mb-4">
        返回列表
      </Button>

      {/* 基础信息 */}
      <Card title="基础信息" className="mb-6">
        <Descriptions column={{ xs: 1, sm: 2, md: 3 }} bordered>
          <Descriptions.Item label="物料编码">{inv.material_code}</Descriptions.Item>
          <Descriptions.Item label="物料名称">{inv.material_name}</Descriptions.Item>
          <Descriptions.Item label="分类">{CATEGORY_MAP[inv.category] || inv.category}</Descriptions.Item>
          <Descriptions.Item label="规格型号">{inv.specification || '-'}</Descriptions.Item>
          <Descriptions.Item label="单位">{inv.unit}</Descriptions.Item>
          <Descriptions.Item label="库存数量">
            <span className={inv.warning ? 'text-red-500 font-bold' : ''}>
              {inv.warning && <WarningFilled className="mr-1" />}
              {inv.quantity}
            </span>
          </Descriptions.Item>
          <Descriptions.Item label="安全库存">{inv.safety_stock}</Descriptions.Item>
          <Descriptions.Item label="仓库位置">{inv.location || '-'}</Descriptions.Item>
          <Descriptions.Item label="状态">
            <Tag color={inv.status === 1 ? 'green' : 'default'}>{getStatusText(inv.status)}</Tag>
          </Descriptions.Item>
          <Descriptions.Item label="备注" span={3}>{inv.remark || '-'}</Descriptions.Item>
          <Descriptions.Item label="创建人">{inv.creator_name || '-'}</Descriptions.Item>
          <Descriptions.Item label="创建时间">{formatDate(inv.created_at)}</Descriptions.Item>
          <Descriptions.Item label="更新时间">{formatDate(inv.updated_at)}</Descriptions.Item>
        </Descriptions>
      </Card>

      {/* 图片区 */}
      {images.length > 0 && (
        <Card title={`图片 (${images.length})`} className="mb-6">
          <Image.PreviewGroup>
            <div className="flex flex-wrap gap-4">
              {images.map(img => (
                <Image
                  key={img.id}
                  width={160}
                  height={160}
                  src={attachmentService.getPreviewUrl(img.id, true)}
                  preview={{ src: attachmentService.getPreviewUrl(img.id) }}
                  className="object-cover rounded border"
                />
              ))}
            </div>
          </Image.PreviewGroup>
        </Card>
      )}

      {/* 文件区 */}
      {documents.length > 0 && (
        <Card title={`文件 (${documents.length})`}>
          <Table
            dataSource={documents}
            rowKey="id"
            pagination={false}
            columns={[
              { title: '文件名', dataIndex: 'display_name' },
              { title: '大小', dataIndex: 'file_size', render: formatFileSize, width: 100 },
              { title: '上传时间', dataIndex: 'created_at', render: formatDate, width: 170 },
              {
                title: '操作', width: 80,
                render: (_, record: Attachment) => (
                  <a href={attachmentService.getDownloadUrl(record.id)} target="_blank">
                    <DownloadOutlined /> 下载
                  </a>
                ),
              },
            ]}
          />
        </Card>
      )}
    </div>
  )
}
```
