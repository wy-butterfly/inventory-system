## 9. 控制器层（Handler）

> Handler 负责：接收请求参数 → 调用 Service → 返回响应。不包含业务逻辑。

### 9.1 `internal/handler/auth_handler.go`

```go
package handler

import (
	"inventory-system/internal/dto/request"
	"inventory-system/internal/service"
	"inventory-system/pkg/errcode"
	apiResp "inventory-system/pkg/response"

	"github.com/gin-gonic/gin"
)

// AuthHandler 认证相关接口
type AuthHandler struct {
	authService *service.AuthService
}

func NewAuthHandler() *AuthHandler {
	return &AuthHandler{
		authService: service.NewAuthService(),
	}
}

// Login 用户登录
// POST /api/v1/auth/login
// 请求体: {"username": "admin", "password": "admin123"}
// 响应: {"code": 0, "data": {"token": "xxx", "user": {...}}}
func (h *AuthHandler) Login(c *gin.Context) {
	// 1. 解析请求参数
	var req request.LoginReq
	if err := c.ShouldBindJSON(&req); err != nil {
		apiResp.Error(c, errcode.ErrInvalidParam.WithMessage("请输入用户名和密码"))
		return
	}

	// 2. 调用 Service 处理登录逻辑
	ip := c.ClientIP()
	token, user, errCode := h.authService.Login(req.Username, req.Password, ip)
	if errCode != nil {
		apiResp.Error(c, errCode)
		return
	}

	// 3. 返回成功响应
	apiResp.Success(c, gin.H{
		"token": token,
		"user": gin.H{
			"id":           user.ID,
			"username":     user.Username,
			"display_name": user.DisplayName,
			"role":         user.Role,
		},
	})
}

// GetMe 获取当前登录用户信息
// GET /api/v1/auth/me
func (h *AuthHandler) GetMe(c *gin.Context) {
	// 从上下文中获取用户ID（由JWT中间件写入）
	userID := c.GetUint("user_id")

	user, errCode := h.authService.GetCurrentUser(userID)
	if errCode != nil {
		apiResp.Error(c, errCode)
		return
	}

	apiResp.Success(c, gin.H{
		"id":           user.ID,
		"username":     user.Username,
		"display_name": user.DisplayName,
		"role":         user.Role,
		"status":       user.Status,
	})
}

// Logout 用户登出
// POST /api/v1/auth/logout
func (h *AuthHandler) Logout(c *gin.Context) {
	// JWT是无状态的，登出只需前端清除Token即可
	// 如果需要服务端主动失效Token，可以维护一个Token黑名单（Redis）
	apiResp.SuccessWithMessage(c, "登出成功", nil)
}
```

```go
// ListOperationLogs 查询操作日志
// GET /api/v1/logs/operations?action=create&start_time=2026-01-01&end_time=2026-12-31&page=1&page_size=20
func (h *LogHandler) ListOperationLogs(c *gin.Context) {
	action := c.Query("action")
	targetType := c.Query("target_type")

	page := 1
	pageSize := 20
	if p := c.Query("page"); p != "" {
		if v, err := time.Parse("2006-01-02", p); err == nil {
			_ = v
		}
	}

	// 解析分页
	if p := c.DefaultQuery("page", "1"); p != "" {
		if v, err := parseInt(p); err == nil {
			page = v
		}
	}
	if ps := c.DefaultQuery("page_size", "20"); ps != "" {
		if v, err := parseInt(ps); err == nil {
			pageSize = v
		}
	}

	// 解析时间范围
	var startTime, endTime *time.Time
	if st := c.Query("start_time"); st != "" {
		if t, err := time.Parse("2006-01-02", st); err == nil {
			startTime = &t
		}
	}
	if et := c.Query("end_time"); et != "" {
		if t, err := time.Parse("2006-01-02", et); err == nil {
			t = t.Add(24*time.Hour - time.Second) // 包含当天
			endTime = &t
		}
	}

	offset := (page - 1) * pageSize
	logs, total, err := h.logRepo.ListOperationLogs(action, targetType, nil, startTime, endTime, offset, pageSize)
	if err != nil {
		apiResp.Error(c, errcode.ErrDatabase)
		return
	}

	apiResp.PageSuccess(c, logs, total, page, pageSize)
}

// ListChangeLogs 查询变更日志
// GET /api/v1/logs/changes?inventory_id=1&page=1&page_size=20
func (h *LogHandler) ListChangeLogs(c *gin.Context) {
	page := 1
	pageSize := 20

	if p := c.DefaultQuery("page", "1"); p != "" {
		if v, err := parseInt(p); err == nil {
			page = v
		}
	}

			pageSize = v
		}
	}

	var inventoryID *uint
	if idStr := c.Query("inventory_id"); idStr != "" {
		if v, err := parseInt(idStr); err == nil {
			id := uint(v)
			inventoryID = &id
		}
	}

	var startTime, endTime *time.Time
	if st := c.Query("start_time"); st != "" {
		if t, err := time.Parse("2006-01-02", st); err == nil {
			startTime = &t
		}
	}
	if et := c.Query("end_time"); et != "" {
		if t, err := time.Parse("2006-01-02", et); err == nil {
			t = t.Add(24*time.Hour - time.Second)
			endTime = &t
		}
	}

	offset := (page - 1) * pageSize
	logs, total, err := h.logRepo.ListChangeLogs(inventoryID, nil, startTime, endTime, offset, pageSize)
	if err != nil {
		apiResp.Error(c, errcode.ErrDatabase)
		return
	}

	apiResp.PageSuccess(c, logs, total, page, pageSize)

// parseInt 辅助：将字符串转为int
func parseInt(s string) (int, error) {
	v, err := time.ParseDuration("0s")
	_ = v
	// 用标准库
	i := 0
	for _, ch := range s {
		if ch < '0' || ch > '9' {
			return 0, errcode.ErrInvalidParam
		}
		i = i*10 + int(ch-'0')
	}
	return i, nil
}
```

```go
package middleware

// CORS中间件：允许前端（不同端口/域名）访问后端API
// 通俗说：浏览器的安全策略默认不允许 localhost:3000 访问 localhost:8080
//         这个中间件告诉浏览器"这个后端允许前端来访问"

import (
	"net/http"

	"github.com/gin-gonic/gin"
)

func CORS() gin.HandlerFunc {
	return func(c *gin.Context) {
		c.Header("Access-Control-Allow-Origin", "*")                                // 允许所有来源（生产环境应改为具体域名）
		c.Header("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS") // 允许的请求方法
		c.Header("Access-Control-Allow-Headers", "Authorization, Content-Type")     // 允许的请求头
		c.Header("Access-Control-Max-Age", "86400")                                 // 预检请求缓存24小时

		// OPTIONS 请求直接返回（浏览器的预检请求）
		if c.Request.Method == http.MethodOptions {
			c.AbortWithStatus(http.StatusNoContent)
			return
		}

		c.Next()
	}
}
```

### 10.2 `internal/middleware/auth.go` — JWT认证

```go
package middleware

// JWT认证中间件：验证请求中的Token，提取用户信息
// 每个需要登录才能访问的接口都会经过这个中间件

import (
	"strings"

	"inventory-system/pkg/errcode"
	"inventory-system/pkg/jwt"
	apiResp "inventory-system/pkg/response"

	"github.com/gin-gonic/gin"
)

func JWTAuth() gin.HandlerFunc {
	return func(c *gin.Context) {
		// 1. 从请求头获取Token
		// 格式：Authorization: Bearer eyJhbGciOiJIUzI1NiIs...
		authHeader := c.GetHeader("Authorization")
		if authHeader == "" {
			apiResp.Error(c, errcode.ErrUnauthorized)
			c.Abort() // 终止请求，不再往下执行
			return
		}

		// 2. 提取Token（去掉 "Bearer " 前缀）
		parts := strings.SplitN(authHeader, " ", 2)
		if len(parts) != 2 || parts[0] != "Bearer" {
			apiResp.Error(c, errcode.ErrUnauthorized)
			c.Abort()
			return
		}
		tokenString := parts[1]

		// 3. 解析并验证Token
		claims, err := jwt.ParseToken(tokenString)
		if err != nil {
			apiResp.Error(c, errcode.ErrUnauthorized)
			c.Abort()
			return
		}

		// 4. 将用户信息写入请求上下文，后续Handler可以通过 c.GetUint("user_id") 获取
		c.Set("user_id", claims.UserID)
		c.Set("username", claims.Username)
		c.Set("role", claims.Role)

		c.Next() // 继续执行下一个中间件或Handler
	}
}
```

### 10.3 `internal/middleware/permission.go` — 权限校验

```go
package middleware

// 权限校验中间件：检查当前用户的角色是否有权限访问当前接口

import (
	"inventory-system/pkg/errcode"
	apiResp "inventory-system/pkg/response"

	"github.com/gin-gonic/gin"
)

// RequireRoles 创建角色检查中间件
// 参数：允许访问的角色列表
// 用法：router.POST("/inventories", RequireRoles("admin", "operator"), handler.Create)
func RequireRoles(roles ...string) gin.HandlerFunc {
	return func(c *gin.Context) {
		// 获取当前用户角色（由JWT中间件写入）
		userRole := c.GetString("role")

		// 检查是否在允许的角色列表中
		allowed := false
		for _, role := range roles {
			if userRole == role {
				allowed = true
				break
			}
		}

		if !allowed {
			apiResp.Error(c, errcode.ErrForbidden)
			c.Abort()
			return
		}

		c.Next()
	}
}

// RequireAdmin 仅管理员可访问
func RequireAdmin() gin.HandlerFunc {
	return RequireRoles("admin")
}
```

### 10.5 `internal/middleware/recovery.go` — 异常恢复

```go
package middleware

// Recovery中间件：捕获程序崩溃（panic），避免整个服务挂掉
// 通俗说：如果某个请求处理过程中代码出错崩溃了，这个中间件会"接住"它，
//         返回500错误响应而不是让整个服务器停止运行

import (
	"fmt"
	"net/http"
	"runtime/debug"

	"inventory-system/pkg/errcode"

	"github.com/gin-gonic/gin"
)

func Recovery() gin.HandlerFunc {
	return func(c *gin.Context) {
		defer func() {
			if err := recover(); err != nil {
				// 打印错误信息和堆栈（方便排查问题）
				fmt.Printf("[PANIC] %v\n%s\n", err, debug.Stack())

				c.JSON(http.StatusInternalServerError, gin.H{
					"code":    errcode.ErrServer.Code,
					"message": "服务器内部错误",
					"data":    nil,
				})
				c.Abort()
			}
		}()
		c.Next()
	}
}
```

```go
package router

// 这个文件负责注册所有API路由
// 通俗说：告诉系统"哪个URL地址对应哪个处理函数"

import (
	"inventory-system/internal/handler"
	"inventory-system/internal/middleware"
	"inventory-system/internal/model"

	"github.com/gin-gonic/gin"
)

// Setup 初始化路由
func Setup(mode string) *gin.Engine {
	// 设置运行模式
	if mode == "release" {
		gin.SetMode(gin.ReleaseMode)
	}

	// 创建Gin引擎实例
	r := gin.New()

	// ========== 注册全局中间件 ==========
	r.Use(middleware.Recovery())      // 异常恢复（必须第一个）
	r.Use(middleware.RequestLogger()) // 请求日志
	r.Use(middleware.CORS())          // 跨域处理

	// ========== 创建Handler实例 ==========
	authHandler := handler.NewAuthHandler()
	inventoryHandler := handler.NewInventoryHandler()
	attachmentHandler := handler.NewAttachmentHandler()
	userHandler := handler.NewUserHandler()
	logHandler := handler.NewLogHandler()
	dashboardHandler := handler.NewDashboardHandler()

	// ========== 健康检查（不需要登录） ==========
	r.GET("/api/health", dashboardHandler.HealthCheck)

	// ========== API v1 路由组 ==========
	v1 := r.Group("/api/v1")

	// ----- 认证接口（不需要登录） -----
	auth := v1.Group("/auth")
	{
		auth.POST("/login", authHandler.Login)
	}

	// ----- 以下接口都需要登录 -----
	v1.Use(middleware.JWTAuth())

	// 认证接口（需要登录）
	{
		v1.POST("/auth/logout", authHandler.Logout)
		v1.GET("/auth/me", authHandler.GetMe)
	}

	// ----- 仪表盘 -----
	{
		v1.GET("/dashboard", dashboardHandler.GetDashboard)
	}

	// ----- 库存接口 -----
	inventories := v1.Group("/inventories")
	{
		// 所有角色都可以查询
		inventories.GET("", inventoryHandler.List)
		inventories.GET("/:id", inventoryHandler.GetByID)
		inventories.GET("/export", inventoryHandler.Export)

		// 管理员和操作员可以新增、编辑
		inventories.POST("",
			middleware.RequireRoles(model.RoleAdmin, model.RoleOperator),
			inventoryHandler.Create)
		inventories.PUT("/:id",
			middleware.RequireRoles(model.RoleAdmin, model.RoleOperator),
			inventoryHandler.Update)

		// 删除（根据需求，操作员删除权限可配置，这里暂时只允许管理员）
		inventories.DELETE("/:id",
			middleware.RequireAdmin(),
			inventoryHandler.Delete)
		inventories.POST("/batch-delete",
			middleware.RequireAdmin(),
			inventoryHandler.BatchDelete)
		inventories.PUT("/:id/restore",
			middleware.RequireAdmin(),
			inventoryHandler.Restore)

		// 附件接口（挂在库存路由下）
		inventories.POST("/:id/attachments",
			middleware.RequireRoles(model.RoleAdmin, model.RoleOperator),
			attachmentHandler.Upload)
		inventories.GET("/:id/attachments", attachmentHandler.List)
	}

	// ----- 附件操作接口 -----
	attachments := v1.Group("/attachments")
	{
		attachments.GET("/:id/download", attachmentHandler.Download)
		attachments.GET("/:id/preview", attachmentHandler.Preview)
		attachments.PUT("/:id",
			middleware.RequireRoles(model.RoleAdmin, model.RoleOperator),
			attachmentHandler.Rename)
		attachments.DELETE("/:id",
			middleware.RequireAdmin(),
			attachmentHandler.Delete)
	}

	// ----- 用户管理接口（仅管理员） -----
	users := v1.Group("/users", middleware.RequireAdmin())
	{
		users.GET("", userHandler.List)
		users.POST("", userHandler.Create)
		users.PUT("/:id", userHandler.Update)
		users.PUT("/:id/reset-password", userHandler.ResetPassword)
	}

	// ----- 日志接口（仅管理员） -----
	logs := v1.Group("/logs", middleware.RequireAdmin())
	{
		logs.GET("/operations", logHandler.ListOperationLogs)
		logs.GET("/changes", logHandler.ListChangeLogs)
	}

	return r
}
```

```go
package main

// 这是整个后端程序的入口文件
// 运行 `go run main.go` 时，Go会从这里的 main() 函数开始执行

import (
	"fmt"
	"os"
	"os/signal"
	"syscall"

	"inventory-system/config"
	"inventory-system/internal/router"
	"inventory-system/pkg/database"
)

func main() {
	// 1. 加载配置文件
	fmt.Println("⏳ 正在加载配置...")
	if err := config.Load("."); err != nil {
		fmt.Printf("❌ 加载配置失败: %v\n", err)
		os.Exit(1) // 退出程序
	}
	fmt.Println("✅ 配置加载成功")

	// 2. 初始化数据库连接
	fmt.Println("⏳ 正在连接数据库...")
	if err := database.Init(); err != nil {
		fmt.Printf("❌ 数据库连接失败: %v\n", err)
		os.Exit(1)
	}
	defer database.Close() // 程序退出时关闭数据库连接

	// 3. 初始化路由
	r := router.Setup(config.GlobalConfig.Server.Mode)

	// 4. 启动HTTP服务器
	port := config.GlobalConfig.Server.Port
	fmt.Printf("🚀 服务器启动成功，监听端口: %d\n", port)
	fmt.Printf("   API地址: http://localhost:%d/api/v1\n", port)
	fmt.Printf("   健康检查: http://localhost:%d/api/health\n", port)
	fmt.Println("   按 Ctrl+C 停止服务器")

	// 在后台启动HTTP服务
	go func() {
		if err := r.Run(fmt.Sprintf(":%d", port)); err != nil {
			fmt.Printf("❌ 服务器启动失败: %v\n", err)
			os.Exit(1)
		}
	}()

	// 5. 优雅退出：监听系统中断信号（Ctrl+C）
	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	<-quit
	fmt.Println("\n🛑 服务器正在关闭...")
}
```

```bash
# ========== 前提条件 ==========
# 已安装 Go 1.21+、MySQL 8.0+

# 看到 "🚀 服务器启动成功" 表示成功
```

### 15.2 常见问题排查

| 问题 | 可能原因 | 解决方法 |
|------|---------|---------|
| 数据库连接失败 | 密码错误或MySQL未启动 | 检查 config.yaml 中的密码；`mysql -u root -p` 测试连接 |
| Token无效 | JWT密钥不一致 | 确认 config.yaml 中 jwt.secret 没有被修改 |
| 文件上传失败 | uploads目录无写权限 | `chmod 755 uploads` 或创建目录 `mkdir -p uploads` |
| 跨域错误 | 前端地址未在CORS中配置 | 开发环境下 CORS 设为 `*`（已配置） |
| 编译报错 | 依赖未下载 | 执行 `go mod download` |
