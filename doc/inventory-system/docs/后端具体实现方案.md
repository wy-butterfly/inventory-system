## 2. 配置管理

### `config/config.go`

```go
package config

// 这个文件负责读取 config.yaml 配置文件中的所有设置
// 通俗说：就像餐厅的"设置面板"，所有可调节的参数都在这里管理

import (
	"fmt"
	"github.com/spf13/viper"
)

// ========== 配置结构体定义 ==========

// Config 是整个应用的配置根结构
// 每个字段对应 config.yaml 中的一个配置块
type Config struct {
	Server   ServerConfig   `mapstructure:"server"`   // 服务器配置
	Database DatabaseConfig `mapstructure:"database"` // 数据库配置
	JWT      JWTConfig      `mapstructure:"jwt"`      // JWT认证配置
	Upload   UploadConfig   `mapstructure:"upload"`   // 文件上传配置
	Log      LogConfig      `mapstructure:"log"`      // 日志配置
}

// ServerConfig 服务器相关配置
type ServerConfig struct {
	Port int    `mapstructure:"port"` // 监听端口，如 8080
	Mode string `mapstructure:"mode"` // 运行模式：debug（开发）或 release（生产）
}

// DatabaseConfig 数据库连接配置
type DatabaseConfig struct {
	Host         string `mapstructure:"host"`           // 数据库地址，如 localhost
	Port         int    `mapstructure:"port"`           // 数据库端口，如 3306
	Name         string `mapstructure:"name"`           // 数据库名称，如 inventory_db
	User         string `mapstructure:"user"`           // 数据库用户名
	Password     string `mapstructure:"password"`       // 数据库密码
	MaxOpenConns int    `mapstructure:"max_open_conns"` // 最大连接数
	MaxIdleConns int    `mapstructure:"max_idle_conns"` // 最大空闲连接数
}

// JWTConfig JWT认证配置
type JWTConfig struct {
	Secret      string `mapstructure:"secret"`       // JWT签名密钥（一串随机字符串）
	ExpireHours int    `mapstructure:"expire_hours"`  // Token过期时间（小时）
}

// UploadConfig 文件上传配置
type UploadConfig struct {
	MaxSize         int64    `mapstructure:"max_size"`          // 单文件最大大小（字节）
	AllowedImageExt []string `mapstructure:"allowed_image_ext"` // 允许的图片格式
	AllowedFileExt  []string `mapstructure:"allowed_file_ext"`  // 允许的文件格式
	StoragePath     string   `mapstructure:"storage_path"`      // 文件存储路径
	ThumbnailWidth  int      `mapstructure:"thumbnail_width"`   // 缩略图宽度（像素）
}

// LogConfig 日志配置
type LogConfig struct {
	Level  string `mapstructure:"level"`  // 日志级别：debug/info/warn/error
	Output string `mapstructure:"output"` // 输出方式：console/file
	Path   string `mapstructure:"path"`   // 日志文件路径
}

// ========== 全局配置变量 ==========

// GlobalConfig 全局配置实例，其他文件通过 config.GlobalConfig 来读取配置
var GlobalConfig *Config

// ========== 加载配置函数 ==========

// Load 读取并解析配置文件
// 参数 path：配置文件的路径，如 "." 表示当前目录
func Load(path string) error {
	// viper 是一个配置管理工具，帮我们读取 yaml 文件
	viper.SetConfigName("config")  // 配置文件名（不含扩展名）
	viper.SetConfigType("yaml")    // 配置文件类型
	viper.AddConfigPath(path)      // 配置文件所在目录

	// 设置默认值（如果配置文件中没有写，就用这些默认值）
	viper.SetDefault("server.port", 8080)
	viper.SetDefault("server.mode", "debug")
	viper.SetDefault("jwt.expire_hours", 24)
	viper.SetDefault("upload.max_size", 20971520)    // 20MB
	viper.SetDefault("upload.storage_path", "./uploads")
	viper.SetDefault("upload.thumbnail_width", 200)
	viper.SetDefault("database.max_open_conns", 100)
	viper.SetDefault("database.max_idle_conns", 10)
	viper.SetDefault("log.level", "info")
	viper.SetDefault("log.output", "console")

	// 读取配置文件
	if err := viper.ReadInConfig(); err != nil {
		return fmt.Errorf("读取配置文件失败: %w", err)
	}

	// 将配置文件的内容解析到 GlobalConfig 结构体中
	GlobalConfig = &Config{}
	if err := viper.Unmarshal(GlobalConfig); err != nil {
		return fmt.Errorf("解析配置文件失败: %w", err)
	}

	return nil
}

// GetDSN 生成数据库连接字符串
// DSN = Data Source Name，是连接数据库所需的完整地址字符串
func (c *DatabaseConfig) GetDSN() string {
	return fmt.Sprintf("%s:%s@tcp(%s:%d)/%s?charset=utf8mb4&parseTime=True&loc=Local",
		c.User,
		c.Password,
		c.Host,
		c.Port,
		c.Name,
	)
}
```

## 3. 数据库连接

### `pkg/database/mysql.go`

```go
package database

// 这个文件负责建立与MySQL数据库的连接
// 通俗说：相当于"拿到仓库的钥匙"，之后所有数据存取操作都通过这个连接进行

import (
	"fmt"
	"time"

	"inventory-system/config"

	"gorm.io/driver/mysql"
	"gorm.io/gorm"
	"gorm.io/gorm/logger"
)

// DB 全局数据库连接实例，其他文件通过 database.DB 来操作数据库
var DB *gorm.DB

// Init 初始化数据库连接
func Init() error {
	cfg := config.GlobalConfig.Database

	// 根据运行模式选择日志级别
	// debug模式下打印所有SQL语句（方便调试），release模式下只打印错误
	var logLevel logger.LogLevel
	if config.GlobalConfig.Server.Mode == "debug" {
		logLevel = logger.Info // 打印所有SQL
	} else {
		logLevel = logger.Error // 只打印错误
	}

	// 建立数据库连接
	var err error
	DB, err = gorm.Open(mysql.Open(cfg.GetDSN()), &gorm.Config{
		Logger: logger.Default.LogMode(logLevel),
	})
	if err != nil {
		return fmt.Errorf("连接数据库失败: %w", err)
	}

	// 获取底层的 sql.DB 对象来设置连接池参数
	sqlDB, err := DB.DB()
	if err != nil {
		return fmt.Errorf("获取数据库实例失败: %w", err)
	}

	// 连接池配置
	sqlDB.SetMaxOpenConns(cfg.MaxOpenConns)        // 最大连接数
	sqlDB.SetMaxIdleConns(cfg.MaxIdleConns)        // 最大空闲连接数
	sqlDB.SetConnMaxLifetime(time.Hour)            // 连接最大存活时间

	// 测试连接是否正常
	if err := sqlDB.Ping(); err != nil {
		return fmt.Errorf("数据库连接测试失败: %w", err)
	}

	fmt.Println("✅ 数据库连接成功")
	return nil
}

// Close 关闭数据库连接（程序退出时调用）
func Close() error {
	sqlDB, err := DB.DB()
	if err != nil {
		return err
	}
	return sqlDB.Close()
}
```

### 4.2 `internal/model/user.go` — 用户模型

```go
package model

// User 用户表对应的Go结构体
// 每个字段后面的 `gorm:"..."` 标签告诉GORM如何映射到数据库列
// `json:"..."` 标签告诉Go如何把这个结构体转换成JSON
type User struct {
	BaseModel               // 嵌入基础字段（ID、CreatedAt、UpdatedAt）
	Username     string `gorm:"type:varchar(50);uniqueIndex;not null" json:"username"`       // 用户名，唯一
	PasswordHash string `gorm:"type:varchar(200);not null" json:"-"`                         // 密码哈希，json:"-" 表示不输出到JSON（安全考虑）
	DisplayName  string `gorm:"type:varchar(100);default:''" json:"display_name"`             // 显示名称
	Role         string `gorm:"type:varchar(20);not null;default:'viewer'" json:"role"`       // 角色：admin/operator/viewer
	Status       int8   `gorm:"type:tinyint;not null;default:1" json:"status"`                // 状态：1=启用，0=停用
}

// TableName 指定表名（GORM默认会自动把结构体名变成复数作为表名）
func (User) TableName() string {
	return "users"
}

// 角色常量定义
const (
	RoleAdmin    = "admin"    // 管理员
	RoleOperator = "operator" // 操作员
	RoleViewer   = "viewer"   // 只读用户
)

// IsAdmin 判断用户是否是管理员
func (u *User) IsAdmin() bool {
	return u.Role == RoleAdmin
}

// IsOperator 判断用户是否是操作员
func (u *User) IsOperator() bool {
	return u.Role == RoleOperator
}

// HasWritePermission 判断用户是否有写入权限（管理员或操作员）
func (u *User) HasWritePermission() bool {
	return u.Role == RoleAdmin || u.Role == RoleOperator
}
```

```go
	// 非数据库字段：用于展示创建人名称（不映射到数据库列）
	CreatorName string `gorm:"-" json:"creator_name,omitempty"`

	// 非数据库字段：库存预警标记
	Warning bool `gorm:"-" json:"warning"`
}

func (Inventory) TableName() string {
	return "inventories"
}

// 分类枚举常量
const (
	CategoryRawMaterial    = "raw_material"     // 原料
	CategoryFinishedProduct = "finished_product" // 成品
	CategorySparePart      = "spare_part"       // 备件
	CategoryOther          = "other"            // 其他
)

// ValidCategories 所有合法的分类值
var ValidCategories = []string{
	CategoryRawMaterial,
	CategoryFinishedProduct,
	CategorySparePart,
	CategoryOther,
}

// IsValidCategory 检查分类值是否合法
func IsValidCategory(category string) bool {
	for _, c := range ValidCategories {
		if c == category {
			return true
		}
	}
	return false
}
```

### 5.1 `internal/dto/request/common_req.go` — 通用请求参数

### 6.3 `pkg/jwt/jwt.go` — JWT工具

```go
package jwt

// 这个文件负责JWT Token的生成和解析
// 通俗说：生成"通行证"和验证"通行证"是否有效

import (
	"errors"
	"time"

	"inventory-system/config"

	jwtLib "github.com/golang-jwt/jwt/v5"
)

// Claims 自定义的JWT载荷（Token里存储的信息）
type Claims struct {
	UserID   uint   `json:"user_id"`  // 用户ID
	Username string `json:"username"` // 用户名
	Role     string `json:"role"`     // 角色
	jwtLib.RegisteredClaims            // JWT标准字段（过期时间等）
}

// GenerateToken 生成JWT Token
// 参数：用户ID、用户名、角色
// 返回：Token字符串、错误
func GenerateToken(userID uint, username, role string) (string, error) {
	cfg := config.GlobalConfig.JWT

	// 设置Token中的信息
	claims := Claims{
		UserID:   userID,
		Username: username,
		Role:     role,
		RegisteredClaims: jwtLib.RegisteredClaims{
			// Token过期时间 = 当前时间 + 配置的过期小时数
			ExpiresAt: jwtLib.NewNumericDate(time.Now().Add(
				time.Duration(cfg.ExpireHours) * time.Hour,
			)),
			IssuedAt: jwtLib.NewNumericDate(time.Now()), // 签发时间
			Issuer:   "inventory-system",                // 签发者
		},
	}

	// 使用密钥生成Token
	token := jwtLib.NewWithClaims(jwtLib.SigningMethodHS256, claims)
	return token.SignedString([]byte(cfg.Secret))
}

// ParseToken 解析并验证JWT Token
// 参数：Token字符串
// 返回：解析出的用户信息、错误
func ParseToken(tokenString string) (*Claims, error) {
	cfg := config.GlobalConfig.JWT

	// 解析Token
	token, err := jwtLib.ParseWithClaims(tokenString, &Claims{},
		func(token *jwtLib.Token) (interface{}, error) {
			return []byte(cfg.Secret), nil
		},
	)

	if err != nil {
		return nil, err
	}

	// 验证Token有效性并提取Claims
	if claims, ok := token.Claims.(*Claims); ok && token.Valid {
		return claims, nil
	}

	return nil, errors.New("无效的Token")
}
```

```go
package hash

// 这个文件负责密码的加密和验证
// 通俗说：把用户的明文密码变成一串乱码存到数据库，登录时对比乱码是否匹配

import "golang.org/x/crypto/bcrypt"

// HashPassword 对密码进行加密
// 输入："admin123" → 输出："$2a$10$N9qo8uLOickgx2ZMRZoMye..."（一串不可逆的哈希值）
func HashPassword(password string) (string, error) {
	// bcrypt.DefaultCost = 10，数字越大越安全但越慢
	bytes, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
	return string(bytes), err
}

// CheckPassword 验证密码是否正确
// 对比明文密码和数据库中存储的哈希值
func CheckPassword(password, hash string) bool {
	err := bcrypt.CompareHashAndPassword([]byte(hash), []byte(password))
	return err == nil
}
```

```go
package upload

// 这个文件负责文件上传相关的校验和处理
// 通俗说：检查文件格式、大小是否合规，生成不冲突的存储路径

import (
	"fmt"
	"image"
	"image/jpeg"
	"image/png"
	"io"
	"mime/multipart"
	"net/http"
	"os"
	"path/filepath"
	"strings"
	"time"

	"inventory-system/config"
	"inventory-system/internal/model"

	"github.com/google/uuid"
	"github.com/nfnt/resize"
)

// ValidateFile 校验上传的文件（格式+大小）
// 返回：文件类型（image/document）、错误
func ValidateFile(file *multipart.FileHeader) (string, error) {
	cfg := config.GlobalConfig.Upload

	// 1. 检查文件大小
	if file.Size > cfg.MaxSize {
		return "", fmt.Errorf("文件大小超过限制（最大%dMB）", cfg.MaxSize/1024/1024)
	}

	// 2. 获取文件扩展名（转小写）
	ext := strings.ToLower(filepath.Ext(file.Filename))

	// 3. 检查是否是合法的图片格式
	for _, allowedExt := range cfg.AllowedImageExt {
		if ext == allowedExt {
			return model.FileTypeImage, nil
		}
	}

	// 4. 检查是否是合法的文件格式
	for _, allowedExt := range cfg.AllowedFileExt {
		if ext == allowedExt {
			return model.FileTypeDocument, nil
		}
	}

	return "", fmt.Errorf("不支持的文件格式: %s", ext)
}

// ValidateMIME 校验文件的MIME类型（第二道安全检查）
func ValidateMIME(file multipart.File) (string, error) {
	// 读取文件前512字节来检测真实类型
	buffer := make([]byte, 512)
	_, err := file.Read(buffer)
	if err != nil {
		return "", err
	}

	// 重置文件读取位置到开头
	file.Seek(0, io.SeekStart)

	// 检测MIME类型
	mimeType := http.DetectContentType(buffer)
	return mimeType, nil
}

// GenerateFilePath 生成文件存储路径
// 格式：uploads/202603/42/a1b2c3d4-e5f6.jpg
func GenerateFilePath(inventoryID uint, originalName string) (string, string) {
	cfg := config.GlobalConfig.Upload
	ext := strings.ToLower(filepath.Ext(originalName))
	fileName := uuid.New().String() + ext
	yearMonth := time.Now().Format("200601")
	dir := filepath.Join(cfg.StoragePath, yearMonth, fmt.Sprintf("%d", inventoryID))
	fullPath := filepath.Join(dir, fileName)
	return dir, fullPath
}

// SaveFile 保存上传的文件到磁盘
func SaveFile(file *multipart.FileHeader, dir, fullPath string) error {
	// 创建存储目录（如果不存在）
	if err := os.MkdirAll(dir, 0755); err != nil {
		return fmt.Errorf("创建目录失败: %w", err)
	}

	// 打开上传的文件
	src, err := file.Open()
	if err != nil {
		return fmt.Errorf("打开上传文件失败: %w", err)
	}
	defer src.Close()

	// 创建目标文件
	dst, err := os.Create(fullPath)
	if err != nil {
		return fmt.Errorf("创建文件失败: %w", err)
	}
	defer dst.Close()

	// 将内容从上传文件复制到目标文件
	if _, err := io.Copy(dst, src); err != nil {
		return fmt.Errorf("写入文件失败: %w", err)
	}

	return nil
}

// GenerateThumbnail 生成图片缩略图
// 返回缩略图路径
func GenerateThumbnail(originalPath string) (string, error) {
	cfg := config.GlobalConfig.Upload

	// 打开原图
	file, err := os.Open(originalPath)
	if err != nil {
		return "", err
	}
	defer file.Close()

	// 解码图片
	img, format, err := image.Decode(file)
	if err != nil {
		return "", err
	}

	// 生成缩略图（保持宽高比，宽度缩放到配置值）
	thumbnail := resize.Resize(uint(cfg.ThumbnailWidth), 0, img, resize.Lanczos3)

	// 生成缩略图路径：在原文件名前加 "thumb_"
	dir := filepath.Dir(originalPath)
	base := filepath.Base(originalPath)
	thumbPath := filepath.Join(dir, "thumb_"+base)

	// 保存缩略图
	thumbFile, err := os.Create(thumbPath)
	if err != nil {
		return "", err
	}
	defer thumbFile.Close()

	// 根据原图格式保存
	switch format {
	case "jpeg":
		err = jpeg.Encode(thumbFile, thumbnail, &jpeg.Options{Quality: 80})
	case "png":
		err = png.Encode(thumbFile, thumbnail)
	default:
		err = jpeg.Encode(thumbFile, thumbnail, &jpeg.Options{Quality: 80})
	}

	if err != nil {
		return "", err
	}

	return thumbPath, nil
}
```

```go
	query := database.DB.Model(&model.User{})

	// 关键词搜索

			"%"+keyword+"%", "%"+keyword+"%")
```

```go
// List 分页查询库存列表

	var total int64

	query := database.DB.Model(&model.Inventory{})

	// 是否显示已删除记录
	if showDeleted {
		query = query.Where("is_deleted = 1")
	} else {
		query = query.Where("is_deleted = 0")
	}

	// 关键词搜索（物料编码或名称模糊匹配）
	if keyword != "" {
		query = query.Where("(material_code LIKE ? OR material_name LIKE ?)",
			"%"+keyword+"%", "%"+keyword+"%")
	}
	// 分类筛选
	if category != "" {
		query = query.Where("category = ?", category)
	}
	// 状态筛选
	if status != nil {
		query = query.Where("status = ?", *status)
	}
	// 库存数量区间
	if minQty != nil {
		query = query.Where("quantity >= ?", *minQty)
	}
	if maxQty != nil {
		query = query.Where("quantity <= ?", *maxQty)
	}
	// 位置筛选
	if location != "" {
		query = query.Where("location LIKE ?", "%"+location+"%")
	}

	// 查询总数
	if err := query.Count(&total).Error; err != nil {
		return nil, 0, err
	}

	// 查询分页数据（按更新时间倒序）
	err := query.Order("updated_at DESC").
		Offset(offset).Limit(limit).
		Find(&inventories).Error

	return inventories, total, err
```

```go
// 返回：JWT Token、用户信息、错误
func (s *AuthService) Login(username, password, ip string) (string, *model.User, *errcode.ErrCode) {
	// 1. 查找用户
	user, err := s.userRepo.FindByUsername(username)
	if err != nil {
		if err == gorm.ErrRecordNotFound {
			return "", nil, errcode.ErrLoginFailed // 用户不存在（统一提示"用户名或密码错误"）

	}

	if !hash.CheckPassword(password, user.PasswordHash) {
		return "", nil, errcode.ErrLoginFailed
	}

	// 4. 生成JWT Token
	token, err := jwt.GenerateToken(user.ID, user.Username, user.Role)
	if err != nil {
		return "", nil, errcode.ErrServer
	}

	// 5. 记录登录日志

		UserID:      user.ID,
		UserName:    user.Username,
		Action:      model.ActionLogin,
		TargetType:  model.TargetUser,

		Description: "用户登录",
		IPAddress:   ip,
	})
```

```go
type InventoryService struct {

	attachmentRepo *repository.AttachmentRepo
	logRepo        *repository.LogRepo
	userRepo       *repository.UserRepo
}

func NewInventoryService() *InventoryService {
	return &InventoryService{

		logRepo:        repository.NewLogRepo(),

// Create 新增库存记录
func (s *InventoryService) Create(req *request.CreateInventoryReq, operatorID uint, operatorName string) (*model.Inventory, *errcode.ErrCode) {
	// 1. 校验分类是否合法
	if !model.IsValidCategory(req.Category) {
		return nil, errcode.ErrInvalidParam.WithMessage("无效的分类值")
	}

	// 2. 校验物料编码唯一性
	existing, err := s.inventoryRepo.FindByMaterialCode(req.MaterialCode)
	if err == nil && existing != nil {
		return nil, errcode.ErrDuplicate.WithMessage("物料编码已存在")
	}
	if err != nil && err != gorm.ErrRecordNotFound {
		return nil, errcode.ErrDatabase
	}

	// 3. 组装数据模型
	status := int8(1) // 默认启用
	if req.Status != nil {
		status = *req.Status
	}

	inv := &model.Inventory{
		MaterialCode:  req.MaterialCode,
		MaterialName:  req.MaterialName,
		Category:      req.Category,
		Specification: req.Specification,
		Unit:          req.Unit,
		Quantity:      req.Quantity,
		SafetyStock:   req.SafetyStock,
		Location:      req.Location,
		Status:        status,
		Remark:        req.Remark,
		Version:       1,
		CreatedBy:     operatorID,
	}

	// 4. 保存到数据库
	if err := s.inventoryRepo.Create(inv); err != nil {
		return nil, errcode.ErrDatabase
	}

	// 5. 记录操作日志
	s.logRepo.CreateOperationLog(&model.OperationLog{
		UserID:      operatorID,
		UserName:    operatorName,
		Action:      model.ActionCreate,
		TargetType:  model.TargetInventory,
		TargetID:    &inv.ID,
		Description: fmt.Sprintf("新增库存: %s (%s)", inv.MaterialName, inv.MaterialCode),
	})

	return inv, nil
}

// GetByID 获取库存详情
func (s *InventoryService) GetByID(id uint) (*model.Inventory, *errcode.ErrCode) {
	inv, err := s.inventoryRepo.FindByID(id)
	if err != nil {
		if err == gorm.ErrRecordNotFound {
			return nil, errcode.ErrNotFound
		}
		return nil, errcode.ErrDatabase
	}

	// 标记预警
	if inv.SafetyStock > 0 && inv.Quantity < inv.SafetyStock {
		inv.Warning = true
	}

	// 查询附件
	attachments, _ := s.attachmentRepo.ListByInventoryID(id)
	inv.Attachments = attachments

	// 查询创建人名称
	nameMap, _ := s.userRepo.GetNamesByIDs([]uint{inv.CreatedBy})
	if name, ok := nameMap[inv.CreatedBy]; ok {
		inv.CreatorName = name
	}

	return inv, nil
}

// Update 编辑库存记录
func (s *InventoryService) Update(id uint, req *request.UpdateInventoryReq, operatorID uint, operatorName string) *errcode.ErrCode {
	// 1. 查找原记录
	oldInv, err := s.inventoryRepo.FindByID(id)
	if err != nil {
		if err == gorm.ErrRecordNotFound {
			return errcode.ErrNotFound
		}
		return errcode.ErrDatabase
	}

	// 2. 准备变更日志（对比新旧值）
	var changeLogs []model.ChangeLog

	// 辅助函数：记录字段变更
	addChange := func(field, oldVal, newVal string) {
		if oldVal != newVal {
			label := model.FieldLabelMap[field]
			if label == "" {
				label = field
			}
			changeLogs = append(changeLogs, model.ChangeLog{
				InventoryID: id,
				UserID:      operatorID,
				UserName:    operatorName,
				FieldName:   field,
				FieldLabel:  label,
				OldValue:    oldVal,
				NewValue:    newVal,
			})
		}
	}

	// 3. 逐字段更新（只更新传入的字段）
	if req.MaterialName != nil {
		addChange("material_name", oldInv.MaterialName, *req.MaterialName)
		oldInv.MaterialName = *req.MaterialName
	}
	if req.Category != nil {
		if !model.IsValidCategory(*req.Category) {
			return errcode.ErrInvalidParam.WithMessage("无效的分类值")
		}
		addChange("category", oldInv.Category, *req.Category)
		oldInv.Category = *req.Category
	}
	if req.Specification != nil {
		addChange("specification", oldInv.Specification, *req.Specification)
		oldInv.Specification = *req.Specification
	}
	if req.Unit != nil {
		addChange("unit", oldInv.Unit, *req.Unit)
		oldInv.Unit = *req.Unit
	}
	if req.Quantity != nil {
		addChange("quantity",
			strconv.FormatFloat(oldInv.Quantity, 'f', 2, 64),
			strconv.FormatFloat(*req.Quantity, 'f', 2, 64))
		oldInv.Quantity = *req.Quantity
	}
	if req.SafetyStock != nil {
		addChange("safety_stock",
			strconv.FormatFloat(oldInv.SafetyStock, 'f', 2, 64),
			strconv.FormatFloat(*req.SafetyStock, 'f', 2, 64))
		oldInv.SafetyStock = *req.SafetyStock
	}
	if req.Location != nil {
		addChange("location", oldInv.Location, *req.Location)
		oldInv.Location = *req.Location
	}
	if req.Status != nil {
		addChange("status",
			strconv.Itoa(int(oldInv.Status)),
			strconv.Itoa(int(*req.Status)))
		oldInv.Status = *req.Status
	}
	if req.Remark != nil {
		addChange("remark", oldInv.Remark, *req.Remark)
		oldInv.Remark = *req.Remark
	}

	oldInv.UpdatedBy = &operatorID

	// 4. 执行更新（带乐观锁）
	if err := s.inventoryRepo.Update(oldInv, req.Version); err != nil {
		if err == gorm.ErrRecordNotFound {
			return errcode.ErrOptimisticLock // 乐观锁冲突
		}
		return errcode.ErrDatabase
	}

	// 5. 保存变更日志
	if len(changeLogs) > 0 {
		s.logRepo.CreateChangeLogs(changeLogs)
	}

	// 6. 记录操作日志
	s.logRepo.CreateOperationLog(&model.OperationLog{
		UserID:      operatorID,
		UserName:    operatorName,
		Action:      model.ActionUpdate,
		TargetType:  model.TargetInventory,
		TargetID:    &id,
		Description: fmt.Sprintf("编辑库存: %s，变更%d个字段", oldInv.MaterialCode, len(changeLogs)),
	})

	return nil
}

// Delete 软删除库存记录
func (s *InventoryService) Delete(id uint, operatorID uint, operatorName string) *errcode.ErrCode {
	inv, err := s.inventoryRepo.FindByID(id)
	if err != nil {
		if err == gorm.ErrRecordNotFound {
			return errcode.ErrNotFound
		}
		return errcode.ErrDatabase
	}

	// 软删除库存
	if err := s.inventoryRepo.SoftDelete(id, operatorID); err != nil {
		return errcode.ErrDatabase
	}

	// 同步软删除附件
	s.attachmentRepo.SoftDeleteByInventoryID(id)

	// 记录日志
	s.logRepo.CreateOperationLog(&model.OperationLog{
		UserID:      operatorID,
		UserName:    operatorName,
		Action:      model.ActionDelete,
		TargetType:  model.TargetInventory,
		TargetID:    &id,
		Description: fmt.Sprintf("删除库存: %s (%s)", inv.MaterialName, inv.MaterialCode),
	})

	return nil
}

// BatchDelete 批量软删除
func (s *InventoryService) BatchDelete(ids []uint, operatorID uint, operatorName string) *errcode.ErrCode {
	if err := s.inventoryRepo.BatchSoftDelete(ids, operatorID); err != nil {
		return errcode.ErrDatabase
	}

	// 同步删除附件
	for _, id := range ids {
		s.attachmentRepo.SoftDeleteByInventoryID(id)
	}

	s.logRepo.CreateOperationLog(&model.OperationLog{
		UserID:      operatorID,
		UserName:    operatorName,
		Action:      model.ActionDelete,
		TargetType:  model.TargetInventory,
		Description: fmt.Sprintf("批量删除%d条库存记录", len(ids)),
	})

	return nil
}

// Restore 恢复已删除的记录
func (s *InventoryService) Restore(id uint, operatorID uint, operatorName string) *errcode.ErrCode {
	if err := s.inventoryRepo.Restore(id); err != nil {
		return errcode.ErrDatabase
	}

	s.logRepo.CreateOperationLog(&model.OperationLog{
		UserID:      operatorID,
		UserName:    operatorName,
		Action:      model.ActionRestore,
		TargetType:  model.TargetInventory,
		TargetID:    &id,
		Description: fmt.Sprintf("恢复库存记录 ID=%d", id),
	})

	return nil
}

// List 分页查询库存列表
func (s *InventoryService) List(query *request.InventoryQuery) ([]model.Inventory, int64, *errcode.ErrCode) {
	inventories, total, err := s.inventoryRepo.List(
		query.Keyword, query.Category, query.Status,
		query.MinQuantity, query.MaxQuantity, query.Location,
		query.ShowDeleted,
		query.GetOffset(), query.GetPageSize(),
	)
	if err != nil {
		return nil, 0, errcode.ErrDatabase
	}

	// 标记预警 + 填充创建人名称
	var creatorIDs []uint
	for i := range inventories {
		if inventories[i].SafetyStock > 0 && inventories[i].Quantity < inventories[i].SafetyStock {
			inventories[i].Warning = true
		}
		creatorIDs = append(creatorIDs, inventories[i].CreatedBy)
	}

	if len(creatorIDs) > 0 {
		nameMap, _ := s.userRepo.GetNamesByIDs(creatorIDs)
		for i := range inventories {
			if name, ok := nameMap[inventories[i].CreatedBy]; ok {
				inventories[i].CreatorName = name
			}
		}
	}

	return inventories, total, nil
}
```

```go
// Upload 上传附件
func (s *AttachmentService) Upload(inventoryID uint, file *multipart.FileHeader,
	operatorID uint, operatorName string) (*model.Attachment, *errcode.ErrCode) {

	// 1. 确认库存记录存在
	_, err := s.inventoryRepo.FindByID(inventoryID)
	if err != nil {
		if err == gorm.ErrRecordNotFound {
			return nil, errcode.ErrNotFound.WithMessage("库存记录不存在")
		}
		return nil, errcode.ErrDatabase
	}

	// 2. 校验文件（格式+大小）
	fileType, err := upload.ValidateFile(file)
	if err != nil {
		return nil, errcode.ErrFileFormat.WithMessage(err.Error())
	}

	// 3. 校验MIME类型
	src, err := file.Open()
	if err != nil {
		return nil, errcode.ErrFileUpload
	}
	mimeType, err := upload.ValidateMIME(src)
	src.Close()
	if err != nil {
		return nil, errcode.ErrFileUpload
	}

	// 4. 生成存储路径
	dir, fullPath := upload.GenerateFilePath(inventoryID, file.Filename)

	// 5. 保存文件到磁盘
	if err := upload.SaveFile(file, dir, fullPath); err != nil {
		return nil, errcode.ErrFileUpload.WithMessage(err.Error())
	}

	// 6. 如果是图片，生成缩略图
	var thumbnailPath string
	if fileType == model.FileTypeImage {
		thumbPath, err := upload.GenerateThumbnail(fullPath)
		if err == nil {
			thumbnailPath = thumbPath
		}
	}

	// 7. 保存附件记录到数据库
	ext := strings.ToLower(filepath.Ext(file.Filename))
	att := &model.Attachment{
		InventoryID:   inventoryID,
		FileName:      file.Filename,
		DisplayName:   file.Filename,
		FilePath:      fullPath,
		FileSize:      file.Size,
		FileType:      fileType,
		MimeType:      mimeType,
		Extension:     ext,
		ThumbnailPath: thumbnailPath,
		UploadedBy:    operatorID,
	}

	if err := s.attachmentRepo.Create(att); err != nil {
		return nil, errcode.ErrDatabase
	}

	// 8. 记录操作日志
	s.logRepo.CreateOperationLog(&model.OperationLog{
		UserID:      operatorID,
		UserName:    operatorName,
		Action:      model.ActionUpload,
		TargetType:  model.TargetAttachment,
		TargetID:    &att.ID,
		Description: fmt.Sprintf("上传文件: %s (库存ID=%d)", file.Filename, inventoryID),
	})

	return att, nil
}
```

```go
// Create 创建用户
func (s *UserService) Create(req *request.CreateUserReq, operatorID uint, operatorName string) (*model.User, *errcode.ErrCode) {
	// 检查用户名唯一性
	existing, err := s.userRepo.FindByUsername(req.Username)
	if err == nil && existing != nil {
		return nil, errcode.ErrDuplicate.WithMessage("用户名已存在")
	}
	if err != nil && err != gorm.ErrRecordNotFound {
		return nil, errcode.ErrDatabase
	}

	// 加密密码
	passwordHash, err := hash.HashPassword(req.Password)
	if err != nil {
		return nil, errcode.ErrServer
	}

	user := &model.User{
		Username:     req.Username,
		PasswordHash: passwordHash,
		DisplayName:  req.DisplayName,
		Role:         req.Role,
		Status:       1,
	}

	if err := s.userRepo.Create(user); err != nil {
		return nil, errcode.ErrDatabase
	}

	s.logRepo.CreateOperationLog(&model.OperationLog{
		UserID:      operatorID,
		UserName:    operatorName,
		Action:      model.ActionCreate,
		TargetType:  model.TargetUser,
		TargetID:    &user.ID,
		Description: fmt.Sprintf("创建用户: %s (角色: %s)", user.Username, user.Role),
	})

	return user, nil
```

```go
// ExportInventories 导出库存数据为Excel
func (s *ExportService) ExportInventories(query *request.InventoryQuery,
	operatorID uint, operatorName string) (*excelize.File, *errcode.ErrCode) {

	// 1. 查询全部数据（不分页）
	inventories, err := s.inventoryRepo.ListAll(
		query.Keyword, query.Category, query.Status,
		query.MinQuantity, query.MaxQuantity,
	)
	if err != nil {
		return nil, errcode.ErrDatabase
	}

	// 2. 创建Excel文件
	f := excelize.NewFile()
	sheetName := "库存数据"
	f.SetSheetName("Sheet1", sheetName)

	// 3. 写入表头
	headers := []string{"物料编码", "物料名称", "分类", "规格型号", "单位",
		"库存数量", "安全库存", "仓库位置", "状态", "备注", "创建时间", "更新时间"}
	for i, h := range headers {
		cell := fmt.Sprintf("%c1", 'A'+i)
		f.SetCellValue(sheetName, cell, h)
	}

	// 4. 写入数据行
	categoryMap := map[string]string{
		"raw_material": "原料", "finished_product": "成品",
		"spare_part": "备件", "other": "其他",
	}
	for i, inv := range inventories {
		row := i + 2
		f.SetCellValue(sheetName, fmt.Sprintf("A%d", row), inv.MaterialCode)
		f.SetCellValue(sheetName, fmt.Sprintf("B%d", row), inv.MaterialName)
		catName := categoryMap[inv.Category]
		if catName == "" {
			catName = inv.Category
		}
		f.SetCellValue(sheetName, fmt.Sprintf("C%d", row), catName)
		f.SetCellValue(sheetName, fmt.Sprintf("D%d", row), inv.Specification)
		f.SetCellValue(sheetName, fmt.Sprintf("E%d", row), inv.Unit)
		f.SetCellValue(sheetName, fmt.Sprintf("F%d", row), strconv.FormatFloat(inv.Quantity, 'f', 2, 64))
		f.SetCellValue(sheetName, fmt.Sprintf("G%d", row), strconv.FormatFloat(inv.SafetyStock, 'f', 2, 64))
		f.SetCellValue(sheetName, fmt.Sprintf("H%d", row), inv.Location)
		statusText := "启用"
		if inv.Status == 0 {
			statusText = "停用"
		}
		f.SetCellValue(sheetName, fmt.Sprintf("I%d", row), statusText)
		f.SetCellValue(sheetName, fmt.Sprintf("J%d", row), inv.Remark)
		f.SetCellValue(sheetName, fmt.Sprintf("K%d", row), inv.CreatedAt.Format("2006-01-02 15:04:05"))
		f.SetCellValue(sheetName, fmt.Sprintf("L%d", row), inv.UpdatedAt.Format("2006-01-02 15:04:05"))
	}

	// 5. 记录导出日志
	s.logRepo.CreateOperationLog(&model.OperationLog{
		UserID:      operatorID,
		UserName:    operatorName,
		Action:      model.ActionExport,
		TargetType:  model.TargetInventory,
		Description: fmt.Sprintf("导出库存数据 %d 条", len(inventories)),
	})

	return f, nil
}
```
